// Conservative source scanner, not a filename convention. JS syntax is parsed
// so test descriptions/regex literals are data, while actual calls/imports are
// checked even in test/check/mutate files. PS here-string C# remains scannable.
import ts from 'typescript';
import {readFileSync,readdirSync,lstatSync} from 'node:fs';
import {resolve,relative,extname,join,dirname} from 'node:path';
import {fileURLToPath} from 'node:url';
const here=dirname(fileURLToPath(import.meta.url));
export const nativeOwnerPaths=Object.freeze([resolve(here,'owned-process.ps1'),resolve(here,'native-launcher.mjs')]);
const extensions=new Set(['.ps1','.psm1','.mjs','.js','.cjs','.cs','.cmd','.bat']);
const forbiddenCall=/^(?:kill|spawn(?:sync)?|execfile(?:sync)?|exec(?:sync)?|fork|terminateprocess)$/i;
export function scanSource(source,file) {
 const hits=[];const report=(at,rule)=>hits.push({file,line:source.slice(0,at).split('\n').length,rule});
 if(/\.(?:mjs|js|cjs)$/i.test(file)) {
  const ast=ts.createSourceFile(file,source,ts.ScriptTarget.Latest,true,ts.ScriptKind.JS);
  function visit(node) {
   if(ts.isCallExpression(node)) {
    const e=node.expression;
    const name=ts.isPropertyAccessExpression(e)?e.name.text:ts.isElementAccessExpression(e)&&ts.isStringLiteral(e.argumentExpression)?e.argumentExpression.text:ts.isIdentifier(e)?e.text:'';
    if(forbiddenCall.test(name))report(node.getStart(ast),'unowned native call');
   }
   // Importing the process API (including aliases, require or dynamic import)
   // is itself forbidden outside the exact launcher path.
   if(ts.isStringLiteral(node)&&/^(?:node:)?child_process$/i.test(node.text))report(node.getStart(ast),'native process API outside owner');
   ts.forEachChild(node,visit);
  }
  visit(ast);
  if(ast.parseDiagnostics.length)report(0,'unparseable JavaScript');
  return hits;
 }
 // Remove comments without discarding executable C# embedded in PowerShell.
 const code=source.replace(/\/\*[\s\S]*?\*\//g,m=>m.replace(/[^\n]/g,' ')).replace(/(^|\n)\s*(?:#|rem\b|::)[^\n]*/gi,m=>m.replace(/[^\n]/g,' ')).replace(/\/\/[^\n]*/g,m=>m.replace(/[^\n]/g,' ')).replace(/`\r?\n/g,'  ');
 const patterns=[
  ['process termination',/\bkill\s*\(|\bStop-Process\b|\b(?:taskkill|tskill)(?:\.exe)?\b|\bTerminateProcess\b/gi],
  ['WMIC termination',/\bwmic(?:\.exe)?\b[^\n;]*(?:\bdelete\b|\bterminate\b)/gi],
  ['CIM termination',/\bInvoke-CimMethod\b[^\n;]*\b(?:-MethodName\s+['"]?Terminate\b|Terminate\b)/gi],
  ['Win32 process termination',/\bWin32_Process\b[^\n;]*\bTerminate\b/gi],
  ['native launch',/\b(?:Start-Process|ProcessStartInfo)\b|\b(?:Process|Diagnostics\.Process)\s*\]?\s*(?:::|\.)\s*Start\s*\(|\b(?:spawn|execFile|exec|fork)(?:Sync)?\s*\(/gi],
  ['direct executable',/(?:^|[\n;|&])\s*(?:&\s*)?["']?(?:[A-Z]:\\[^\n"']*?|[\w.\\/-]+)\.exe["']?(?=\s|$)/gi],
 ];
 for(const [rule,pattern] of patterns)for(const match of code.matchAll(pattern))report(match.index,rule);
 return hits;
}
export function scanWindowsTree(root=new URL('.',import.meta.url)) {
 root=resolve(root instanceof URL?fileURLToPath(root):root);
 const hits=[];
 function walk(dir) {
  for(const entry of readdirSync(dir,{withFileTypes:true})) {
   const path=join(dir,entry.name);
   if(entry.isSymbolicLink()){hits.push({file:relative(root,path),line:1,rule:'source symlink refused'});continue;}
   if(entry.isDirectory()){walk(path);continue;}
   if(!extensions.has(extname(path).toLowerCase())||nativeOwnerPaths.includes(path))continue;
   if(!lstatSync(path).isFile())continue;
   hits.push(...scanSource(readFileSync(path,'utf8'),relative(root,path)));
  }
 }
 walk(root);return hits;
}
if(process.argv[1]&&resolve(process.argv[1])===fileURLToPath(import.meta.url)) {
 const hits=scanWindowsTree();for(const hit of hits)console.error(`${hit.file}:${hit.line}: ${hit.rule}`);
 console.log(`Native ownership scanner: ${hits.length} violations`);process.exitCode=hits.length?1:0;
}
