function run(name,cwd,args,status) {
  const result=spawnSync(process.execPath,args,{cwd,encoding:'utf8'});
  const output=result.stdout+result.stderr;
  writeFileSync(join(raw,name+'.log'),output);
  writeFileSync(join(report,'afix-r3-'+name+'.log'),output.replace(/[ \t]+$/gm,''));
  assert.equal(result.status,status,output);
  if(status===1) {
    assert.match(output,/AssertionError \[ERR_ASSERTION\]/);
    assert.doesNotMatch(output,/ERR_MODULE_NOT_FOUND|MODULE_NOT_FOUND/);
  }
  console.log(`${name}: expected exit ${status}`);
}
